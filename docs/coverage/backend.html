// Command server starts the calculator REST API.
//
// Configuration (environment variables):
//
//        PORT             port to listen on (default 8080)
//        ALLOWED_ORIGINS  comma-separated CORS origins (default http://localhost:5173)
//        STATIC_DIR       if set, also serves the built frontend from this directory
package main

import (
        "context"
        "errors"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "strings"
        "syscall"
        "time"

        "github.com/marciosalsa/sezzle-calculator/backend/internal/api"
)

func main() {
        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

        port := getenv("PORT", "8080")
        origins := strings.Split(getenv("ALLOWED_ORIGINS", "http://localhost:5173"), ",")
        staticDir := os.Getenv("STATIC_DIR")

        var handler http.Handler = api.NewHandler(logger).Routes()
        if staticDir != "" {
                handler = api.WithStaticFiles(handler, staticDir)
        }
        handler = api.Chain(handler, api.Recover(logger), api.Logging(logger), api.CORS(origins))

        srv := &http.Server{
                Addr:              ":" + port,
                Handler:           handler,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        go func() {
                logger.Info("server listening", "addr", srv.Addr, "static_dir", staticDir)
                if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                        logger.Error("server failed", "err", err)
                        os.Exit(1)
                }
        }()

        <-ctx.Done()
        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                logger.Error("graceful shutdown failed", "err", err)
        }
}

func getenv(key, fallback string) string {
        if v := os.Getenv(key); v != "" {
                return v
        }
        return fallback
}

// Package api exposes the calculator domain over a JSON REST interface.
package api

import (
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "strings"

        "github.com/marciosalsa/sezzle-calculator/backend/internal/calculator"
)

// maxBodyBytes caps request bodies; a calculation payload is tiny.
const maxBodyBytes = 1 << 10 // 1 KiB

// Error codes returned in the "error.code" field. They are stable and meant
// to be switched on by clients, while "error.message" is human readable.
const (
        CodeInvalidJSON      = "INVALID_JSON"
        CodeInvalidOperand   = "INVALID_OPERAND"
        CodeUnknownOperation = "UNKNOWN_OPERATION"
        CodeDivisionByZero   = "DIVISION_BY_ZERO"
        CodeNegativeSqrt     = "NEGATIVE_SQUARE_ROOT"
        CodeResultOutOfRange = "RESULT_OUT_OF_RANGE"
        CodePayloadTooLarge  = "PAYLOAD_TOO_LARGE"
        CodeUnsupportedMedia = "UNSUPPORTED_MEDIA_TYPE"
        CodeInternal         = "INTERNAL_ERROR"
)

// CalculateRequest is the body of POST /api/v1/{operation}.
//
// Pointers are used so that a missing field can be told apart from an
// explicit zero (e.g. {"a": 0, "b": 0} is valid, {"a": 0} is not for "add").
type CalculateRequest struct {
        A *float64 `json:"a"`
        B *float64 `json:"b,omitempty"`
}

// CalculateResponse is returned on success.
type CalculateResponse struct {
        Operation calculator.Operation `json:"operation"`
        Operands  []float64            `json:"operands"`
        Result    float64              `json:"result"`
}

// ErrorBody is the payload of every non-2xx response.
type ErrorBody struct {
        Error ErrorDetail `json:"error"`
}

// ErrorDetail carries a machine-readable code and a human-readable message.
type ErrorDetail struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// Handler wires HTTP routes to the calculator.
type Handler struct {
        logger *slog.Logger
}

// NewHandler builds a Handler. A nil logger discards logs.
func NewHandler(logger *slog.Logger) *Handler {
        if logger == nil {
                logger = slog.New(slog.NewTextHandler(io.Discard, nil))
        }
        return &Handler{logger: logger}
}

// Routes registers all API routes on a new ServeMux.
func (h *Handler) Routes() *http.ServeMux {
        mux := http.NewServeMux()
        mux.HandleFunc("GET /healthz", h.health)
        mux.HandleFunc("GET /api/v1/operations", h.listOperations)
        mux.HandleFunc("POST /api/v1/{operation}", h.calculate)
        return mux
}

func (h *Handler) health(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func (h *Handler) listOperations(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]any{"operations": calculator.Operations()})
}

func (h *Handler) calculate(w http.ResponseWriter, r *http.Request) {
        op := calculator.Operation(r.PathValue("operation"))

        arity, err := calculator.ArityOf(op)
        if err != nil {
                writeError(w, http.StatusNotFound, CodeUnknownOperation,
                        fmt.Sprintf("operation %q is not supported", op))
                return
        }

        if ct := r.Header.Get("Content-Type"); ct != "" && !strings.HasPrefix(ct, "application/json") {
                writeError(w, http.StatusUnsupportedMediaType, CodeUnsupportedMedia,
                        "Content-Type must be application/json")
                return
        }

        req, status, code, msg := decodeRequest(w, r)
        if code != "" {
                writeError(w, status, code, msg)
                return
        }

        operands, err := operandsFor(arity, req)
        if err != nil {
                writeError(w, http.StatusBadRequest, CodeInvalidOperand, err.Error())
                return
        }

        result, err := calculator.Calculate(op, operands...)
        if err != nil {
                status, code := mapDomainError(err)
                if status == http.StatusInternalServerError {
                        h.logger.Error("unexpected calculator error", "operation", op, "err", err)
                }
                writeError(w, status, code, err.Error())
                return
        }

        writeJSON(w, http.StatusOK, CalculateResponse{Operation: op, Operands: operands, Result: result})
}

// decodeRequest strictly parses the JSON body. On failure it returns a
// non-empty error code together with the HTTP status and message to send.
func decodeRequest(w http.ResponseWriter, r *http.Request) (CalculateRequest, int, string, string) {
        var req CalculateRequest

        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()

        if err := dec.Decode(&req); err != nil {
                var maxErr *http.MaxBytesError
                var typeErr *json.UnmarshalTypeError
                switch {
                case errors.As(err, &maxErr):
                        return req, http.StatusRequestEntityTooLarge, CodePayloadTooLarge, "request body is too large"
                case errors.Is(err, io.EOF):
                        return req, http.StatusBadRequest, CodeInvalidJSON, "request body is empty"
                case errors.As(err, &typeErr) && typeErr.Field == "":
                        return req, http.StatusBadRequest, CodeInvalidJSON, "request body must be a JSON object"
                case errors.As(err, &typeErr):
                        return req, http.StatusBadRequest, CodeInvalidOperand,
                                fmt.Sprintf("field %q must be a number", typeErr.Field)
                default:
                        return req, http.StatusBadRequest, CodeInvalidJSON, "request body must be valid JSON: " + err.Error()
                }
        }
        // Reject trailing data such as `{"a":1}{"a":2}`.
        if dec.More() {
                return req, http.StatusBadRequest, CodeInvalidJSON, "request body must contain a single JSON object"
        }
        return req, 0, "", ""
}

func operandsFor(arity calculator.Arity, req CalculateRequest) ([]float64, error) {
        if req.A == nil {
                return nil, errors.New(`field "a" is required`)
        }
        switch arity {
        case calculator.Unary:
                if req.B != nil {
                        return nil, errors.New(`field "b" is not allowed for this operation`)
                }
                return []float64{*req.A}, nil
        default:
                if req.B == nil {
                        return nil, errors.New(`field "b" is required`)
                }
                return []float64{*req.A, *req.B}, nil
        }
}

func mapDomainError(err error) (int, string) {
        switch {
        case errors.Is(err, calculator.ErrDivisionByZero):
                return http.StatusUnprocessableEntity, CodeDivisionByZero
        case errors.Is(err, calculator.ErrNegativeSqrt):
                return http.StatusUnprocessableEntity, CodeNegativeSqrt
        case errors.Is(err, calculator.ErrNonFiniteResult):
                return http.StatusUnprocessableEntity, CodeResultOutOfRange
        case errors.Is(err, calculator.ErrNonFiniteInput), errors.Is(err, calculator.ErrInvalidOperands):
                return http.StatusBadRequest, CodeInvalidOperand
        case errors.Is(err, calculator.ErrUnknownOperation):
                return http.StatusNotFound, CodeUnknownOperation
        default:
                return http.StatusInternalServerError, CodeInternal
        }
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, ErrorBody{Error: ErrorDetail{Code: code, Message: message}})
}

// Middleware wraps an http.Handler.
type Middleware func(http.Handler) http.Handler

// Chain applies middlewares so that the first one listed is the outermost.
func Chain(h http.Handler, mws ...Middleware) http.Handler {
        for i := len(mws) - 1; i >= 0; i-- {
                h = mws[i](h)
        }
        return h
}

// CORS allows browser calls from the given origins ("*" allows any).
// With an empty list it is a no-op, which is the case when the frontend is
// served from the same origin (Docker image).
func CORS(allowedOrigins []string) Middleware {
        allowAll := false
        allowed := make(map[string]bool, len(allowedOrigins))
        for _, o := range allowedOrigins {
                o = strings.TrimSpace(o)
                if o == "*" {
                        allowAll = true
                }
                if o != "" {
                        allowed[o] = true
                }
        }

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        origin := r.Header.Get("Origin")
                        if origin != "" && (allowAll || allowed[origin]) {
                                w.Header().Set("Access-Control-Allow-Origin", origin)
                                w.Header().Set("Vary", "Origin")
                                w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                                w.Header().Set("Access-Control-Allow-Headers", "Content-Type")
                                w.Header().Set("Access-Control-Max-Age", "600")
                        }
                        if r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != "" {
                                w.WriteHeader(http.StatusNoContent)
                                return
                        }
                        next.ServeHTTP(w, r)
                })
        }
}

// Recover turns panics into a 500 JSON response instead of dropping the connection.
func Recover(logger *slog.Logger) Middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                if rec := recover(); rec != nil {
                                        logger.Error("panic recovered", "panic", rec, "path", r.URL.Path)
                                        writeError(w, http.StatusInternalServerError, CodeInternal, "internal server error")
                                }
                        }()
                        next.ServeHTTP(w, r)
                })
        }
}

type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (s *statusRecorder) WriteHeader(code int) {
        s.status = code
        s.ResponseWriter.WriteHeader(code)
}

// Logging emits one structured log line per request.
func Logging(logger *slog.Logger) Middleware {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                        next.ServeHTTP(rec, r)
                        logger.Info("request",
                                "method", r.Method,
                                "path", r.URL.Path,
                                "status", rec.status,
                                "duration", time.Since(start).String(),
                        )
                })
        }
}

// WithStaticFiles serves a single-page app from dir for every request that
// is not handled by the API (paths under /api/ and /healthz). Unknown paths
// fall back to index.html so client-side routing keeps working.
func WithStaticFiles(api http.Handler, dir string) http.Handler {
        files := http.FileServer(http.Dir(dir))
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if strings.HasPrefix(r.URL.Path, "/api/") || r.URL.Path == "/healthz" {
                        api.ServeHTTP(w, r)
                        return
                }
                path := filepath.Join(dir, filepath.Clean("/"+r.URL.Path))
                if info, err := os.Stat(path); err != nil || info.IsDir() {
                        http.ServeFile(w, r, filepath.Join(dir, "index.html"))
                        return
                }
                files.ServeHTTP(w, r)
        })
}

// Package calculator contains the pure arithmetic domain logic.
//
// It has no knowledge of HTTP or JSON so it can be tested in isolation and
// reused by any transport (REST, gRPC, CLI...).
package calculator

import (
        "errors"
        "fmt"
        "math"
        "sort"
)

// Operation identifies a supported arithmetic operation.
type Operation string

// Supported operations.
const (
        Add        Operation = "add"
        Subtract   Operation = "subtract"
        Multiply   Operation = "multiply"
        Divide     Operation = "divide"
        Power      Operation = "power"
        Sqrt       Operation = "sqrt"
        Percentage Operation = "percentage"
)

// Domain errors. Callers should compare with errors.Is.
var (
        ErrUnknownOperation = errors.New("unknown operation")
        ErrInvalidOperands  = errors.New("invalid number of operands")
        ErrDivisionByZero   = errors.New("division by zero")
        ErrNegativeSqrt     = errors.New("square root of a negative number")
        ErrNonFiniteInput   = errors.New("operands must be finite numbers")
        ErrNonFiniteResult  = errors.New("result is not a finite number")
)

// Arity is the number of operands an operation expects.
type Arity int

const (
        Unary  Arity = 1
        Binary Arity = 2
)

type definition struct {
        arity       Arity
        description string
        fn          func(x []float64) (float64, error)
}

var registry = map[Operation]definition{
        Add: {Binary, "a + b", func(x []float64) (float64, error) {
                return x[0] + x[1], nil
        }},
        Subtract: {Binary, "a - b", func(x []float64) (float64, error) {
                return x[0] - x[1], nil
        }},
        Multiply: {Binary, "a × b", func(x []float64) (float64, error) {
                return x[0] * x[1], nil
        }},
        Divide: {Binary, "a ÷ b", func(x []float64) (float64, error) {
                if x[1] == 0 {
                        return 0, ErrDivisionByZero
                }
                return x[0] / x[1], nil
        }},
        Power: {Binary, "a raised to the power of b", func(x []float64) (float64, error) {
                return math.Pow(x[0], x[1]), nil
        }},
        Sqrt: {Unary, "square root of a", func(x []float64) (float64, error) {
                if x[0] < 0 {
                        return 0, ErrNegativeSqrt
                }
                return math.Sqrt(x[0]), nil
        }},
        Percentage: {Binary, "a percent of b (a × b / 100)", func(x []float64) (float64, error) {
                return x[0] * x[1] / 100, nil
        }},
}

// Info describes an operation for discovery endpoints.
type Info struct {
        Name        Operation `json:"name"`
        Arity       Arity     `json:"arity"`
        Description string    `json:"description"`
}

// Operations returns every supported operation sorted by name.
func Operations() []Info {
        out := make([]Info, 0, len(registry))
        for name, def := range registry {
                out = append(out, Info{Name: name, Arity: def.arity, Description: def.description})
        }
        sort.Slice(out, func(i, j int) bool { return out[i].Name < out[j].Name })
        return out
}

// ArityOf returns how many operands op expects.
func ArityOf(op Operation) (Arity, error) {
        def, ok := registry[op]
        if !ok {
                return 0, fmt.Errorf("%w: %q", ErrUnknownOperation, op)
        }
        return def.arity, nil
}

// Calculate executes op with the given operands.
//
// It guarantees that a nil error is only returned together with a finite
// result, so callers never have to deal with NaN or ±Inf (which JSON cannot
// represent anyway).
func Calculate(op Operation, operands ...float64) (float64, error) {
        def, ok := registry[op]
        if !ok {
                return 0, fmt.Errorf("%w: %q", ErrUnknownOperation, op)
        }
        if len(operands) != int(def.arity) {
                return 0, fmt.Errorf("%w: %s expects %d, got %d", ErrInvalidOperands, op, def.arity, len(operands))
        }
        for _, v := range operands {
                if math.IsNaN(v) || math.IsInf(v, 0) {
                        return 0, ErrNonFiniteInput
                }
        }

        result, err := def.fn(operands)
        if err != nil {
                return 0, err
        }
        if math.IsNaN(result) || math.IsInf(result, 0) {
                return 0, ErrNonFiniteResult
        }
        // Normalise negative zero (e.g. -0 * 5) so clients always see 0.
        if result == 0 {
                result = 0
        }
        return result, nil
}
